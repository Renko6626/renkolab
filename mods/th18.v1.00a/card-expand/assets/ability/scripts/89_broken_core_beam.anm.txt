script script89 {
    // 破损核心的闪电：从电球连到敌人的一条线段，**瞬发**，不是飞行物。
    // 伤害不在这里，是 C 建的矩形伤害源（`0x45dfa0`，angle = 瞄准角 / w = 距离 / h = 粗细）。
    // 这里只负责画：C 在起脚本后写 VM 的 pos（电球位置）、rotation.z（瞄准角）、scale.x（距离 / 256）。
    // 所以脚本里**不许**碰 scale / rotate（会被插值器盖掉 C 写的值）；anchor(1, 0) = 左端对齐，
    // 贴图的左端就钉在电球上，向右长到敌人身上。
    // 抖动：C 把 scale.x（距离 / 256）写进 %F0，脚本每帧用随机的 %F1 当 scale.y ——
    // 幅度 0.65–1.35 随机、且逐帧正负交替（锯齿上下翻面），电弧就在两点之间来回抽。
    // 第 0 帧的 scale 由 C 直接写（脚本这时还没跑到 scale），第 1 帧起脚本接管。
-1:
    layer(13);
    sprite(@BROKEN_CORE_BOLT);
    anchor(1, 0);
    type(1);
    blendMode(1);
    zWriteDisable(1);
0:
    alpha(255);
+1: // 1
    fsetMul(%F1, %RANDF, 0.7f);
    fadd(%F1, 0.65f);
    scale(%F0, %F1);
    alpha(255);
+1: // 2
    fsetMul(%F1, %RANDF, -0.7f);
    fadd(%F1, -0.65f);
    scale(%F0, %F1);
    alpha(200);
+1: // 3
    fsetMul(%F1, %RANDF, 0.7f);
    fadd(%F1, 0.65f);
    scale(%F0, %F1);
    alpha(255);
+1: // 4
    fsetMul(%F1, %RANDF, -0.7f);
    fadd(%F1, -0.65f);
    scale(%F0, %F1);
    alpha(190);
+1: // 5
    fsetMul(%F1, %RANDF, 0.7f);
    fadd(%F1, 0.65f);
    scale(%F0, %F1);
    alpha(255);
+1: // 6
    fsetMul(%F1, %RANDF, -0.7f);
    fadd(%F1, -0.65f);
    scale(%F0, %F1);
    alphaTime(3, 0, 0);
+3: // 9
    delete();
}
